% ---------- ATS Helpers ----------
\newcommand{\resumeItem}[1]{\item \normalsize #1}
% One \hfill row per entry, not two: two stacked \hfill lines with short left-hand
% text get read as a two-column table during PDF text extraction, which detaches
% the dates from the employer/school. Location sits beside the org name instead.
\newcommand{\resumeSubheading}[4]{
  \vspace{0.10em}
  \textbf{#1} \textbar\ \textit{#3} \textbar\ #4 \hfill \textbf{#2}\\[0.05em]
}
